```
package com.example.myapplication;

import android.app.Activity;
import android.os.Bundle;
import android.widget.*;

public class MainActivity extends Activity {

    EditText name;
    Button btn;
    TextView result;

        super.onCreate(b);
        setContentView(R.layout.activity_main);

        name = findViewById(R.id.name);

            result.setText("Hello " + s);
```

Output: The application displays a welcome message, accepts a name and displays the entered name in a coloured text.

### 2. MyView\.java

Create a new Java class named `MyView`.

```
package com.example.myapplication;

import android.content.Context;
import android.view.View;
import android.graphics.Canvas;
import android.graphics.Paint;
import android.graphics.Color;

public class MyView extends View {

    Paint p = new Paint();

    public MyView(Context context) {
        super(context);
    }

    @Override
    protected void onDraw(Canvas c) {
        super.onDraw(c);

        p.setColor(Color.RED);
        p.setStrokeWidth(8);
        c.drawLine(50, 50, 300, 50, p);

        p.setColor(Color.BLUE);
        c.drawCircle(150, 180, 60, p);

        p.setColor(Color.GREEN);
        c.drawRect(50, 280, 250, 400, p);

        p.setColor(Color.BLACK);
        p.setTextSize(40);
        c.drawText("Hello Android", 50, 480, p);
    }
}
```

Output: The screen displays a red line, blue circle, green rectangle and text.

Note: No additional permissions are required.

### 2. DBHelper.java

```
package com.example.myapplication;

import android.database.sqlite.*;
import android.content.Context;

public class DBHelper extends SQLiteOpenHelper {

    public DBHelper(Context c) {
        super(c, "TodoDB", null, 1);
    }

    @Override
    public void onCreate(SQLiteDatabase db) {
        db.execSQL(
            "CREATE TABLE tasks " +
            "(id INTEGER PRIMARY KEY AUTOINCREMENT, " +
            "task TEXT)"
        );
    }

    @Override
    public void onUpgrade(SQLiteDatabase db,
                          int oldVersion, int newVersion) {
        db.execSQL("DROP TABLE IF EXISTS tasks");
        onCreate(db);
    }
}
```

### 3. TodoProvider.java

```
package com.example.myapplication;

import android.content.*;
import android.database.*;
import android.database.sqlite.*;

public class TodoProvider extends ContentProvider {

    DBHelper helper;

    public static final String AUTHORITY =
            "com.example.myapplication.provider";

    public static final String URL =
            "content://" + AUTHORITY + "/tasks";

    @Override
    public boolean onCreate() {
        helper = new DBHelper(getContext());
        return true;
    }

    @Override
    public Uri insert(Uri uri, ContentValues values) {
        SQLiteDatabase db = helper.getWritableDatabase();

        long id = db.insert("tasks", null, values);

        return ContentUris.withAppendedId(uri, id);
    }

    @Override
    public Cursor query(Uri uri, String[] projection,
                        String selection, String[] args,
                        String sortOrder) {

        SQLiteDatabase db = helper.getReadableDatabase();

        return db.query("tasks", projection, selection,
                args, null, null, sortOrder);
    }

    @Override
    public int delete(Uri uri, String selection, String[] args) {
        SQLiteDatabase db = helper.getWritableDatabase();
        return db.delete("tasks", selection, args);
    }

    @Override
    public int update(Uri uri, ContentValues values,
                      String selection, String[] args) {
        SQLiteDatabase db = helper.getWritableDatabase();
        return db.update("tasks", values, selection, args);
    }

    @Override
    public String getType(Uri uri) {
        return "vnd.android.cursor.dir/vnd.example.tasks";
    }
}
```

### 4. MainActivity.java

```
package com.example.myapplication;

import android.app.Activity;
import android.os.Bundle;
import android.content.*;
import android.database.Cursor;
import android.net.Uri;
import android.widget.*;

public class MainActivity extends Activity {

    EditText task;
    TextView result;

    Uri uri = Uri.parse(TodoProvider.URL);

    @Override
    protected void onCreate(Bundle b) {
        super.onCreate(b);
        setContentView(R.layout.activity_main);

        task = findViewById(R.id.task);
        result = findViewById(R.id.result);

        Button add = findViewById(R.id.add);
        Button show = findViewById(R.id.show);

        add.setOnClickListener(v -> {
            ContentValues values = new ContentValues();

            values.put("task",
                    task.getText().toString());

            getContentResolver().insert(uri, values);

            task.setText("");
            Toast.makeText(this, "Task Added",
                    Toast.LENGTH_SHORT).show();
        });

        show.setOnClickListener(v -> {
            Cursor c = getContentResolver().query(
                    uri, null, null, null, null);

            StringBuilder data = new StringBuilder();

            if (c != null) {
                while (c.moveToNext()) {
                    data.append(c.getString(
                            c.getColumnIndexOrThrow("task")))
                            .append("\n");
                }
                c.close();
            }

            result.setText(data.toString());
        });
    }
}
```

### 5. AndroidManifest.xml

Add the following provider inside the application tag:

```
<provider
    android:name=".TodoProvider"
    android:authorities="com.example.myapplication.provider"
    android:exported="false"/>
```

Output: Users can enter tasks, save them in the SQLite database and display the saved tasks using Content Provider.

Aim: To develop an Android application using Notification Manager.

Concepts used: NotificationManager, NotificationChannel and NotificationCompat.

### 2. MainActivity.java

```
package com.example.myapplication;

import android.app.*;
import android.os.Bundle;
import android.widget.*;
import android.content.pm.PackageManager;

import androidx.core.app.NotificationCompat;
import androidx.core.app.NotificationManagerCompat;
import androidx.core.content.ContextCompat;

public class MainActivity extends Activity {

    String channelId = "my_channel";

    @Override
    protected void onCreate(Bundle b) {
        super.onCreate(b);
        setContentView(R.layout.activity_main);

        createChannel();

        Button btn = findViewById(R.id.notify);

        btn.setOnClickListener(v -> {

            if (android.os.Build.VERSION.SDK_INT >= 33 &&
                ContextCompat.checkSelfPermission(this,
                "android.permission.POST_NOTIFICATIONS")
                != PackageManager.PERMISSION_GRANTED) {

                requestPermissions(
                    new String[]{
                        "android.permission.POST_NOTIFICATIONS"
                    }, 100);

                return;
            }

            showNotification();
        });
    }

    void createChannel() {
        if (android.os.Build.VERSION.SDK_INT >= 26) {
            NotificationChannel channel =
                new NotificationChannel(
                    channelId,
                    "My Notifications",
                    NotificationManager.IMPORTANCE_DEFAULT);

            NotificationManager nm =
                getSystemService(NotificationManager.class);

            nm.createNotificationChannel(channel);
        }
    }

    void showNotification() {

        NotificationCompat.Builder builder =
            new NotificationCompat.Builder(this, channelId)
            .setSmallIcon(android.R.drawable.ic_dialog_info)
            .setContentTitle("Android Notification")
            .setContentText("Your notification is working!")
            .setPriority(NotificationCompat.PRIORITY_DEFAULT)
            .setAutoCancel(true);

        NotificationManagerCompat.from(this)
            .notify(1, builder.build());
    }

    @Override
    public void onRequestPermissionsResult(
            int requestCode, String[] permissions,
            int[] grantResults) {

        super.onRequestPermissionsResult(
                requestCode, permissions, grantResults);

        if (requestCode == 100 &&
            grantResults.length > 0 &&
            grantResults[0] == PackageManager.PERMISSION_GRANTED) {
            showNotification();
        }
    }
}
```

### 3. AndroidManifest.xml

Add this permission before the application tag:

```

    android:name="android.permission.POST_NOTIFICATIONS"/>
```

Required dependency: Add AndroidX Core to your app-level Gradle dependencies if it is not already present.


Output: Clicking the button displays a notification in the notification panel.

### 2. SmsReceiver.java

Create a new Java class named `SmsReceiver`.

```
package com.example.myapplication;

import android.content.*;
import android.telephony.SmsMessage;
import android.widget.Toast;

public class SmsReceiver extends BroadcastReceiver {

    @Override
    public void onReceive(Context context, Intent intent) {

        Object[] data = (Object[])
                intent.getExtras().get("pdus");

        if (data != null) {
            for (Object pdu : data) {

                SmsMessage sms =
                    SmsMessage.createFromPdu(
                        (byte[]) pdu,
                        intent.getStringExtra("format"));

                String message = sms.getMessageBody();

                Toast.makeText(context,
                    "New Message: " + message,
                    Toast.LENGTH_LONG).show();
            }
        }
    }
}
```

### 3. AndroidManifest.xml

Add the following permission and receiver:

```

<application>

        android:exported="true">

        <intent-filter>
            <action android:name="android.provider.Telephony.SMS_RECEIVED"/>
        </intent-filter>

    </receiver>

</application>
```

Important requirements:

- Grant SMS permission at runtime before testing.
- Use a physical Android device with SMS capability.
- Test only with permission and a legitimate test message.
- SMS reception may be restricted on some devices and Android versions.

Output: When an SMS is received, the application displays an alert message using Toast.

# Experiment 7: Creating, Setting and Cancelling Alarms

Aim: To develop an Android application for creating, setting and cancelling alarms.

Concepts used: AlarmManager, PendingIntent and BroadcastReceiver.

### 1. activity_main.xml

```
        android:layout_width="wrap_content"
        android:layout_height="wrap_content"
        android:text="Set Alarm"/>

    <Button
        android:id="@+id/cancel"
        android:layout_width="wrap_content"
        android:layout_height="wrap_content"
        android:text="Cancel Alarm"/>

</LinearLayout>
```

### 2. MainActivity.java

```
package com.example.myapplication;

import android.app.*;
import android.os.Bundle;
import android.content.*;
import android.widget.*;

public class MainActivity extends Activity {

    AlarmManager alarmManager;
    PendingIntent pendingIntent;

    @Override
    protected void onCreate(Bundle b) {
        super.onCreate(b);
        setContentView(R.layout.activity_main);

        alarmManager =
            (AlarmManager) getSystemService(ALARM_SERVICE);

        Intent intent = new Intent(this, AlarmReceiver.class);

        pendingIntent = PendingIntent.getBroadcast(
            this, 0, intent,
            PendingIntent.FLAG_UPDATE_CURRENT |
            PendingIntent.FLAG_IMMUTABLE);

        Button set = findViewById(R.id.set);
        Button cancel = findViewById(R.id.cancel);

        set.setOnClickListener(v -> {

            long time = System.currentTimeMillis()
                    + 10000;

            alarmManager.set(
                AlarmManager.RTC_WAKEUP,
                time,
                pendingIntent);

            Toast.makeText(this,
                "Alarm Set for 10 Seconds",
                Toast.LENGTH_SHORT).show();
        });

        cancel.setOnClickListener(v -> {

            alarmManager.cancel(pendingIntent);

            Toast.makeText(this,
                "Alarm Cancelled",
                Toast.LENGTH_SHORT).show();
        });
    }
}
```

### 3. AlarmReceiver.java

```
package com.example.myapplication;

import android.content.*;
import android.widget.Toast;

public class AlarmReceiver extends BroadcastReceiver {

    @Override
    public void onReceive(Context context, Intent intent) {

        Toast.makeText(context,
            "Alarm Time Reached!",
            Toast.LENGTH_LONG).show();
    }
}
```

### 4. AndroidManifest.xml

Add the receiver inside the application tag:

```
<receiver
    android:name=".AlarmReceiver"
    android:exported="false"/>
```

Output:

- Set Alarm: Creates an alarm after 10 seconds.
- Cancel Alarm: Cancels the scheduled alarm.
- When the alarm triggers, a message is displayed.

Note: This example uses `AlarmManager.set()` and does not require exact-alarm special access.

```
    <TextView
```

### 2. MainActivity.java

```
package com.example.myapplication;

import android.app.*;
import android.os.Bundle;
import android.location.*;
import android.content.pm.PackageManager;
import android.widget.*;

public class MainActivity extends Activity {

    LocationManager manager;
    TextView result;

    @Override
    protected void onCreate(Bundle b) {
        super.onCreate(b);
        setContentView(R.layout.activity_main);

        result = findViewById(R.id.result);
        Button btn = findViewById(R.id.location);

        manager = (LocationManager)
                getSystemService(LOCATION_SERVICE);

        btn.setOnClickListener(v -> {

            if (checkSelfPermission(
                android.Manifest.permission.ACCESS_FINE_LOCATION)
                != PackageManager.PERMISSION_GRANTED) {

                requestPermissions(
                    new String[]{
                        android.Manifest.permission.ACCESS_FINE_LOCATION
                    }, 100);

                return;
            }

            getLocation();
        });
    }

    void getLocation() {

        try {

            Location loc = manager.getLastKnownLocation(
                    LocationManager.GPS_PROVIDER);

            if (loc != null) {

                double lat = loc.getLatitude();
                double lon = loc.getLongitude();

                result.setText(
                    "Latitude: " + lat +
                    "\nLongitude: " + lon);

            } else {
                result.setText(
                    "Location not available. Try again outdoors.");
            }

        } catch (SecurityException e) {
            result.setText("Location Permission Required");
        }
    }

    @Override
    public void onRequestPermissionsResult(
            int requestCode, String[] permissions,
            int[] grantResults) {

        super.onRequestPermissionsResult(
                requestCode, permissions, grantResults);

        if (requestCode == 100 &&
            grantResults.length > 0 &&
            grantResults[0] == PackageManager.PERMISSION_GRANTED) {

            getLocation();
        }
    }
}
```

### 3. AndroidManifest.xml

Add the permission before the application tag:

```
<uses-permission
    android:name="android.permission.ACCESS_FINE_LOCATION"/>

<uses-permission
    android:name="android.permission.ACCESS_COARSE_LOCATION"/>
```

Output: The application displays the latitude and longitude of the device.

Note: Enable GPS and test on a physical device or an Android emulator with a simulated location. This example reads the last known GPS location, so it may not be immediately available.

## Important Viva Questions

1. What is an Activity?\
   An Activity represents a single screen in an Android application.
2. What is XML?\
   XML is used to design the user interface of Android applications.
3. What is an Event Listener?\
   It is used to handle user actions such as button clicks.
4. What is Canvas?\
   Canvas is used to draw graphical objects on the screen.

   It manages and displays notifications.

   It receives and responds to broadcast messages from the Android system or applications.

   It schedules operations to occur at a specified time.
10. What is GPS?\
    GPS stands for Global Positioning System. It helps determine the geographical location of a device.
